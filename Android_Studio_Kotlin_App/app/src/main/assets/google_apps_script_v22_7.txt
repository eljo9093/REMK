var MAX_DEVICES = 3;

function createJsonResponse(obj, callback) {
  var txt = JSON.stringify(obj);
  if (callback) {
    return ContentService.createTextOutput(callback + "(" + txt + ")")
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(txt)
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  try {
    var params = (e && e.parameter) ? e.parameter : {};
    return handleUnifiedRemkRequest(params, {});
  } catch (err) {
    return createJsonResponse({ valid: false, message: "Serverfehler (GET): " + err.toString() });
  }
}

function doPost(e) {
  try {
    var params = (e && e.parameter) ? e.parameter : {};
    var rawBody = (e && e.postData && e.postData.contents) ? e.postData.contents : "{}";
    var body = {};
    try { body = JSON.parse(rawBody); } catch (err) { body = {}; }
    return handleUnifiedRemkRequest(params, body);
  } catch (err) {
    return createJsonResponse({ valid: false, message: "Serverfehler (POST): " + err.toString() });
  }
}

// Vereinheitlicht den Sync-Kanal zwischen Watch & Smartphone, berücksichtigt aber andere Kunden mit Start-Key "1"
function normalizeSyncBucketKey(rawKey, rawUser) {
  var k = String(rawKey || "").trim().toUpperCase().replace(/\.0$/, "");
  var u = String(rawUser || "").trim().toUpperCase();

  // Wenn ein anderer bekannter Kunde (nicht Jochen/Admin) denselben Start-Key "1" nutzt, bekommt er seinen eigenen Bucket!
  if (u && u !== "ADMIN" && u.indexOf("JOCHEN") === -1) {
    if (k === "1" || !k) {
      return "REMK_SYNC_USER_" + u.replace(/[^A-Z0-9_-]/g, "_");
    }
  }

  // Jochens Master-Sync-Kanal (egal ob auf Handy/Uhr "1", "26011990", "ADMIN", "JOCHEN" oder "REMK-VIP-JOCHEN" steht)
  if (!k || k === "1" || k === "26011990" || k === "REMK-VIP-JOCHEN" || k === "ADMIN" || k === "JOCHEN" || k === "JOCHEN RUBE") {
    return "REMK_SYNC_MASTER_JOCHEN";
  }

  return "REMK_SYNC_" + k.replace(/[^A-Z0-9_-]/g, "_");
}

function mergeWatchAndPhoneState(existingObj, incomingPayload) {
  var incomingObj = {};
  try {
    incomingObj = (typeof incomingPayload === "string") ? JSON.parse(incomingPayload) : (incomingPayload || {});
  } catch (e) {
    incomingObj = {};
  }

  var merged = {};
  for (var k in existingObj) merged[k] = existingObj[k];
  for (var k2 in incomingObj) merged[k2] = incomingObj[k2];

  // 1. Bewahre immer den aktuellsten Trainings-/Ernährungsplan der Smartphone-App für die Uhr
  if (incomingObj.watchPlanCompactJson) {
    merged.watchPlanCompactJson = incomingObj.watchPlanCompactJson;
  } else if (existingObj.watchPlanCompactJson) {
    merged.watchPlanCompactJson = existingObj.watchPlanCompactJson;
  }

  // 2. Höchster Wasserstand des Tages gewinnt verlustfrei
  var oldWater = parseFloat(existingObj.drankLiters || 0) || 0;
  var newWater = parseFloat(incomingObj.drankLiters || 0) || 0;
  merged.drankLiters = Math.max(oldWater, newWater);
  merged.hydrationMl = Math.round(merged.drankLiters * 1000);

  // 3. Höchste erledigte Satz-Anzahl & Satz-Matrix (completedSetsMap) zusammenführen
  var oldSets = parseInt(existingObj.doneSetsToday || 0, 10) || 0;
  var newSets = parseInt(incomingObj.doneSetsToday || 0, 10) || 0;
  merged.doneSetsToday = Math.max(oldSets, newSets);

  try {
    var oldMap = (typeof existingObj.completedSetsMap === "string")
      ? JSON.parse(existingObj.completedSetsMap || "{}")
      : (existingObj.completedSetsMap || {});
    var newMap = (typeof incomingObj.completedSetsMap === "string")
      ? JSON.parse(incomingObj.completedSetsMap || "{}")
      : (incomingObj.completedSetsMap || {});
    for (var mk in newMap) {
      oldMap[mk] = Math.max(parseInt(oldMap[mk] || 0, 10) || 0, parseInt(newMap[mk] || 0, 10) || 0);
    }
    merged.completedSetsMap = JSON.stringify(oldMap);
  } catch (e) {}

  // 4. Event-Queue zusammenführen & bereits vom Smartphone bestätigte ackEventIds bereinigen
  var ackList = Array.isArray(incomingObj.ackEventIds)
    ? incomingObj.ackEventIds
    : (Array.isArray(existingObj.ackEventIds) ? existingObj.ackEventIds : []);
  var ackMap = {};
  for (var a = 0; a < ackList.length; a++) ackMap[String(ackList[a])] = true;
  merged.ackEventIds = ackList.slice(-80);

  var combinedEvents = [];
  var seenIds = {};
  var listA = Array.isArray(existingObj.eventsQueue) ? existingObj.eventsQueue : [];
  var listB = Array.isArray(incomingObj.eventsQueue) ? incomingObj.eventsQueue : [];
  var both = listA.concat(listB);
  for (var i = 0; i < both.length; i++) {
    var ev = both[i];
    if (!ev || !ev.id) continue;
    var eid = String(ev.id);
    if (ackMap[eid] || seenIds[eid]) continue;
    seenIds[eid] = true;
    combinedEvents.push(ev);
  }
  merged.eventsQueue = combinedEvents.slice(-80);

  // 5. Merge logValuesMap (exakte Gewichte x Wdh. zwischen iOS, PC & Android APK)
  try {
    var oldLogs = (typeof existingObj.logValuesMap === "string")
      ? JSON.parse(existingObj.logValuesMap || "{}")
      : (existingObj.logValuesMap || {});
    var newLogs = (typeof incomingObj.logValuesMap === "string")
      ? JSON.parse(incomingObj.logValuesMap || "{}")
      : (incomingObj.logValuesMap || {});
    for (var lk in newLogs) {
      if (newLogs[lk]) oldLogs[lk] = newLogs[lk];
    }
    merged.logValuesMap = JSON.stringify(oldLogs);
  } catch (e) {}

  // 6. Merge seatPinsMap (dauerhafte Geräte- & Sitzhöhen-Pins pro Übung)
  try {
    var oldPins = (typeof existingObj.seatPinsMap === "string")
      ? JSON.parse(existingObj.seatPinsMap || "{}")
      : (existingObj.seatPinsMap || {});
    var newPins = (typeof incomingObj.seatPinsMap === "string")
      ? JSON.parse(incomingObj.seatPinsMap || "{}")
      : (incomingObj.seatPinsMap || {});
    for (var pk in newPins) {
      if (newPins[pk]) oldPins[pk] = newPins[pk];
    }
    merged.seatPinsMap = JSON.stringify(oldPins);
  } catch (e) {}

  // 7. Erzeuge Klartext Coach-Live-Radar (für Spalte O in Google Sheets wie bei Everfit/Trainerize)
  var dSets = merged.doneSetsToday || 0;
  var wLiters = Number(merged.drankLiters || 0).toFixed(2).replace(".", ",");
  var oKcal = merged.openKcal !== undefined ? merged.openKcal : "-";
  var srcLabel = merged.source || "live";
  var timeShort = Utilities.formatDate(new Date(), "Europe/Berlin", "dd.MM. HH:mm");
  merged.coachReadableSummary = "🟢 Aktiv (" + timeShort + ") | ✓ " + dSets + " Sätze heute | 🍎 " + oKcal + " kcal offen | 💧 " + wLiters + " L | " + srcLabel;

  return merged;
}

function getLicenseSheet(ss) {
  return ss.getSheetByName("Lizenzen") || ss.getSheets()[0];
}

function findAthleteRowUnified(data, is11ColLayout, enteredUser, enteredKey) {
  var uLow = String(enteredUser || "").trim().toLowerCase();
  var kLow = String(enteredKey || "").trim().replace(/\.0$/, "").toLowerCase();

  // Master-Admin Erkennung für Jochen (falls Jochen mit "26011990", "REMK-VIP-JOCHEN" oder "Admin" reinkommt)
  var isJochenMasterKey = (kLow === "26011990" || kLow === "remk-vip-jochen" || kLow === "admin");

  if (is11ColLayout) {
    // 1. Durchgang: Name/Nickname UND (Eigenes Passwort ODER Start-Key) passen zusammen
    if (uLow && kLow) {
      for (var i = 1; i < data.length; i++) {
        var rName = String(data[i][0] || "").trim().toLowerCase();
        var rStart = String(data[i][1] || "").trim().replace(/\.0$/, "").toLowerCase();
        var rNick = String(data[i][7] || "").trim().toLowerCase();
        var rPw = String(data[i][9] || "").trim().replace(/\.0$/, "").toLowerCase();
        var uMatch = (rName === uLow) || (rNick === uLow) || (rName.indexOf(uLow) !== -1) || (uLow.indexOf(rName) !== -1 && rName.length >= 3);
        var kMatch = (rPw && rPw === kLow) || (rStart && rStart === kLow);
        if (uMatch && kMatch) return i;
      }
    }
    // 2. Durchgang: Eindeutiges Eigenes Passwort (Spalte J)
    for (var j = 1; j < data.length; j++) {
      var cPw = String(data[j][9] || "").trim().replace(/\.0$/, "").toLowerCase();
      if (cPw && cPw === kLow) return j;
    }
    // 3. Durchgang: Nickname (Spalte H) oder Klarname (Spalte A) wurde direkt als Sync-Key auf der Uhr eingegeben
    for (var n = 1; n < data.length; n++) {
      var nVal = String(data[n][7] || "").trim().toLowerCase();
      var nmVal = String(data[n][0] || "").trim().toLowerCase();
      if ((nVal && nVal === kLow) || (nmVal && nmVal === kLow)) return n;
      if (isJochenMasterKey && (nmVal.indexOf("jochen") !== -1 || nVal === "admin")) return n;
    }
    // 4. Durchgang: Start-Key (Spalte B)
    for (var s = 1; s < data.length; s++) {
      var sVal = String(data[s][1] || "").trim().replace(/\.0$/, "").toLowerCase();
      if (sVal && sVal === kLow) return s;
    }
  } else {
    // 7-Spalten-Layout (A=Key, B=Name, C=Status, D=Rolle, E=Erstellt, F=Notizen, G=Geräte, H=Nickname, J=Wunsch-PW)
    for (var r = 1; r < data.length; r++) {
      var rk = String(data[r][0] || "").trim().replace(/\.0$/, "").toLowerCase();
      var rn = String(data[r][1] || "").trim().toLowerCase();
      var rnick = String(data[r][7] || "").trim().toLowerCase();
      var rcpw = String(data[r][9] || "").trim().replace(/\.0$/, "").toLowerCase();
      if (rk === kLow || (rcpw && rcpw === kLow) || (rnick && rnick === kLow)) {
        if (!uLow || rn.indexOf(uLow) !== -1 || uLow.indexOf(rn) !== -1 || rnick === uLow) {
          return r;
        }
      }
    }
    for (var r2 = 1; r2 < data.length; r2++) {
      var rk2 = String(data[r2][0] || "").trim().replace(/\.0$/, "").toLowerCase();
      if (rk2 === kLow) return r2;
    }
  }
  return -1;
}

function handleUnifiedRemkRequest(params, body) {
  var action = String(body.action || params.action || "verify").trim();
  var callback = String(params.callback || "").trim();

  // 1. KI-Teller-Foto-Analyse (Gemini 2.5 Flash Vision API für Smartphone APK)
  if (action === "analyzePlatePhoto" || action === "analyzeFood") {
    var imgB64 = String(body.imageBase64 || "");
    var hint = String(body.voiceHint || "");
    return createJsonResponse(analyzePlatePhotoWithGemini(imgB64, hint), callback);
  }

  var enteredUser = String(body.user || body.name || params.user || params.name || "").trim();
  var enteredKey = String(body.key || body.licenseKey || params.key || "").trim();
  var newKey = String(body.newKey || params.newKey || "").trim();
  var newNickname = String(body.nickname || params.nickname || "").trim();
  var deviceId = String(body.deviceId || params.deviceId || "UNKNOWN").trim();
  var syncPayload = body.syncData || params.syncData || null;

  // 2. SCHNELLER ECHTZEIT-SYNC (syncPush / syncPull) FÜR WATCH & SMARTPHONE APK
  if (action === "syncPush" || action === "syncPull") {
    var bucket = normalizeSyncBucketKey(enteredKey, enteredUser);
    var props = PropertiesService.getScriptProperties();
    var cache = CacheService.getScriptCache();

    var existingRaw = cache.get(bucket) || props.getProperty(bucket) || "";
    var existingObj = {};
    if (existingRaw) {
      try { existingObj = JSON.parse(existingRaw); } catch (e) { existingObj = {}; }
    }

    if (action === "syncPush" && syncPayload) {
      var merged = mergeWatchAndPhoneState(existingObj, syncPayload);
      var nowIso = new Date().toISOString();
      merged.lastUpdatedTs = Date.now();
      merged.lastSyncIso = nowIso;
      merged.lastDeviceId = deviceId || "UNKNOWN";

      var serialized = JSON.stringify(merged);
      cache.put(bucket, serialized, 21600);
      props.setProperty(bucket, serialized);

      // Schreibe zusätzlich in die Google-Tabelle (Spalte L & M bzw. K & L oder Free_Sync), damit im Sheet alles sichtbar ist
      try {
        var ssSync = SpreadsheetApp.getActiveSpreadsheet();
        if (enteredKey.toUpperCase().indexOf("FREE-") === 0 || enteredKey.toUpperCase() === "FREE_USER") {
          writeFreeSyncRow(ssSync, enteredKey.toUpperCase(), serialized, nowIso);
        } else {
          var shSync = getLicenseSheet(ssSync);
          var dSync = shSync.getDataRange().getValues();
          var is11 = String(dSync[0][0] || "").trim().toLowerCase() === "name";
          var rIdx = findAthleteRowUnified(dSync, is11, enteredUser, enteredKey);
          if (rIdx !== -1) {
            var syncCol = is11 ? 12 : 11; // Spalte L (12) im 11-Spalten-Layout, Spalte K (11) im 7-Spalten-Layout
            if (!dSync[0][syncCol - 1]) shSync.getRange(1, syncCol).setValue("Cloud_Sync_Data");
            if (!dSync[0][syncCol]) shSync.getRange(1, syncCol + 1).setValue("Last_Sync");
            shSync.getRange(rIdx + 1, syncCol).setValue(serialized);
            shSync.getRange(rIdx + 1, syncCol + 1).setValue(nowIso);
            if (is11) {
              if (!dSync[0][14]) shSync.getRange(1, 15).setValue("Coach_Live_Status");
              shSync.getRange(rIdx + 1, 15).setValue(merged.coachReadableSummary || "");
            }
          }
        }
      } catch (sheetErr) {}

      return createJsonResponse({
        valid: true,
        synced: true,
        status: "synced",
        bucket: bucket,
        lastSync: nowIso,
        syncData: merged
      }, callback);
    }

    // action === "syncPull"
    // Falls Cache/Properties noch leer waren, versuche initial aus der Tabelle zu lesen
    if (!existingRaw) {
      try {
        var ssPull = SpreadsheetApp.getActiveSpreadsheet();
        var shPull = getLicenseSheet(ssPull);
        var dPull = shPull.getDataRange().getValues();
        var is11Pull = String(dPull[0][0] || "").trim().toLowerCase() === "name";
        var rPull = findAthleteRowUnified(dPull, is11Pull, enteredUser, enteredKey);
        if (rPull !== -1) {
          var sCol = is11Pull ? 11 : 10; // 0-based Index: 11 = Spalte L, 10 = Spalte K
          var rawCell = String(dPull[rPull][sCol] || "").trim();
          if (rawCell) {
            existingObj = JSON.parse(rawCell);
            cache.put(bucket, rawCell, 21600);
          }
        }
      } catch (e) {}
    }

    return createJsonResponse({
      valid: true,
      bucket: bucket,
      syncData: existingObj
    }, callback);
  }

  // 3. LIZENZ-PRÜFUNG, GERÄTE-VERWALTUNG, PASSWORT, NICKNAME & BACKUP
  if (!enteredKey && !enteredUser) {
    return createJsonResponse({ valid: false, message: "Bitte Lizenz-Key oder Passwort eingeben." }, callback);
  }

  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = getLicenseSheet(ss);
  var data = sheet.getDataRange().getValues();
  var is11ColLayout = (data.length > 0 && String(data[0][0] || "").trim().toLowerCase() === "name");

  var targetRow = findAthleteRowUnified(data, is11ColLayout, enteredUser, enteredKey);

  // Master-Admin Fallback für Jochen, falls die Tabelle leer ist oder ein VIP-Key genutzt wird
  var cleanUpperKey = enteredKey.toUpperCase();
  if (targetRow === -1 && (cleanUpperKey === "26011990" || cleanUpperKey === "REMK-VIP-JOCHEN")) {
    return createJsonResponse({
      valid: true,
      user: "Jochen Rube",
      realName: "Jochen Rube",
      role: "Admin",
      unlimitedDevices: true
    }, callback);
  }

  if (targetRow === -1) {
    return createJsonResponse({ valid: false, message: "Ungültiger Lizenz-Key oder Passwort." }, callback);
  }

  var isWatch = deviceId.toUpperCase().indexOf("WATCH-") === 0;

  if (is11ColLayout) {
    // 11–14 SPALTEN LAYOUT (RE_MK_Athleten_optimiert)
    var headers = sheet.getRange(1, 1, 1, 14).getValues()[0];
    if (!headers[11]) sheet.getRange(1, 12).setValue("Cloud_Sync_Data");
    if (!headers[12]) sheet.getRange(1, 13).setValue("Last_Sync");
    if (!headers[13]) sheet.getRange(1, 14).setValue("Cloud_Backup_JSON");

    var rowNameVal = String(data[targetRow][0] || "Athlet").trim();
    var rowStatus = String(data[targetRow][2] || "AKTIV").trim().toLowerCase();
    var rowRole = String(data[targetRow][3] || "Kunde").trim();
    var dev1 = String(data[targetRow][4] || "").trim();
    var dev2 = String(data[targetRow][5] || "").trim();
    var dev3 = String(data[targetRow][6] || "").trim();
    var rowNickVal = String(data[targetRow][7] || "").trim();
    var rowExpires = String(data[targetRow][8] || "LEBENSLANG").trim();
    var rowEmailLogin = String(data[targetRow][10] || "").trim();
    var rowSyncJson = String(data[targetRow][11] || "").trim();
    var rowLastSync = String(data[targetRow][12] || "").trim();
    var rowBackupJson = String(data[targetRow][13] || "").trim();

    if (rowStatus === "gesperrt" || rowStatus === "inaktiv" || rowStatus === "beendet") {
      return createJsonResponse({ valid: false, locked: true, message: "Zugang ist aktuell pausiert oder gesperrt." }, callback);
    }

    var isAdmin = rowRole.toLowerCase().indexOf("admin") !== -1 ||
                  rowRole.toLowerCase().indexOf("trainer") !== -1 ||
                  rowNameVal.toLowerCase().indexOf("jochen") !== -1;
    var devices = [dev1, dev2, dev3].filter(function(d) { return d && d.length > 0; });

    if (action === "resetDevices") {
      var firstDev = (deviceId && deviceId !== "UNKNOWN" && !isWatch) ? deviceId : "";
      sheet.getRange(targetRow + 1, 5, 1, 3).setValues([[firstDev, "", ""]]);
      SpreadsheetApp.flush();
      return createJsonResponse({ valid: true, user: rowNickVal || rowNameVal, realName: rowNameVal, role: rowRole, message: "Geräteplätze zurückgesetzt!" }, callback);
    }

    if (action === "setFirstNickname") {
      var chosenNick = newNickname || enteredUser;
      if (!chosenNick || chosenNick.length < 3) {
        return createJsonResponse({ valid: false, message: "Bitte wähle einen Nickname mit mindestens 3 Zeichen." }, callback);
      }
      for (var nIdx = 1; nIdx < data.length; nIdx++) {
        if (nIdx === targetRow) continue;
        var otherNick = String(data[nIdx][7] || "").trim().toLowerCase();
        if (otherNick && otherNick === chosenNick.toLowerCase()) {
          return createJsonResponse({ valid: false, message: "Dieser Nickname ist bereits vergeben." }, callback);
        }
      }
      sheet.getRange(targetRow + 1, 8).setValue(chosenNick);
      SpreadsheetApp.flush();
      return createJsonResponse({ valid: true, user: chosenNick, realName: rowNameVal, role: rowRole, message: "Nickname gespeichert!" }, callback);
    }

    if (action === "changePassword" && newKey.length >= 4) {
      sheet.getRange(targetRow + 1, 10).setValue(newKey);
      SpreadsheetApp.flush();
      return createJsonResponse({ valid: true, user: rowNickVal || rowNameVal, realName: rowNameVal, role: rowRole, message: "Passwort gespeichert!" }, callback);
    }

    if (action === "saveBackup" || action === "syncCustomLibraryV202") {
      var bData = body.backup || body.library || params.backup || "";
      var bStr = (typeof bData === "string") ? bData : JSON.stringify(bData);
      sheet.getRange(targetRow + 1, 14).setValue(bStr);
      SpreadsheetApp.flush();
      return createJsonResponse({ valid: true, saved: true }, callback);
    }
    if (action === "loadBackup") {
      return createJsonResponse({ valid: true, backup: rowBackupJson }, callback);
    }

    if (deviceId && deviceId !== "UNKNOWN" && !isWatch) {
      if (devices.indexOf(deviceId) === -1) {
        if (devices.length >= MAX_DEVICES && !isAdmin) {
          return createJsonResponse({ valid: false, locked: true, limitReached: true, message: "Maximal 3 Geräte erreicht. Bitte Geräte zurücksetzen." }, callback);
        }
        if (devices.length >= MAX_DEVICES && isAdmin) devices.shift();
        devices.push(deviceId);
        sheet.getRange(targetRow + 1, 5, 1, 3).setValues([[devices[0] || "", devices[1] || "", devices[2] || ""]]);
      }
    }

    var todayShort = new Date().toISOString().slice(0, 10);
    if (rowEmailLogin.indexOf(todayShort) === -1) {
      var emailPart = rowEmailLogin.split("|")[0].trim();
      var nextColK = (emailPart && emailPart.indexOf("Login:") === -1)
        ? (emailPart + " | Login: " + todayShort)
        : ("Login: " + todayShort);
      sheet.getRange(targetRow + 1, 11).setValue(nextColK);
    }

    var loginSync = null;
    try { loginSync = rowSyncJson ? JSON.parse(rowSyncJson) : null; } catch (e) {}

    return createJsonResponse({
      valid: true,
      user: rowNickVal || rowNameVal,
      realName: rowNameVal,
      role: rowRole,
      expiresAt: rowExpires,
      needsFirstSetup: (!isAdmin && rowNickVal === ""),
      unlimitedDevices: isAdmin,
      syncData: loginSync,
      lastSync: rowLastSync
    }, callback);

  } else {
    // 7-SPALTEN LAYOUT (A=Key, B=Name, C=Status, D=Rolle, E=Erstellt, F=Notizen, G=Geräte)
    var rName7 = String(data[targetRow][1] || "Athlet").trim();
    var rStatus7 = String(data[targetRow][2] || "Aktiv").trim().toLowerCase();
    var rRole7 = String(data[targetRow][3] || "Kunde").trim();
    var rDevsRaw7 = String(data[targetRow][6] || "").trim();
    var rNick7 = String(data[targetRow][7] || "").trim();

    if (rStatus7 === "gesperrt" || rStatus7 === "inaktiv" || rStatus7 === "beendet") {
      return createJsonResponse({ valid: false, locked: true, message: "Zugang gesperrt." }, callback);
    }

    var isAdmin7 = rRole7.toLowerCase().indexOf("admin") !== -1 || rName7.toLowerCase().indexOf("jochen") !== -1;
    var devs7 = rDevsRaw7 ? rDevsRaw7.split(",").map(function(s){ return s.trim(); }).filter(Boolean) : [];

    if (action === "resetDevices") {
      devs7 = (deviceId && deviceId !== "UNKNOWN" && !isWatch) ? [deviceId] : [];
      sheet.getRange(targetRow + 1, 7).setValue(devs7.join(", "));
      SpreadsheetApp.flush();
      return createJsonResponse({ valid: true, user: rNick7 || rName7, role: rRole7, message: "Geräte zurückgesetzt!" }, callback);
    }

    if (action === "changePassword" && newKey.length >= 4) {
      sheet.getRange(targetRow + 1, 1).setValue(newKey);
      SpreadsheetApp.flush();
      return createJsonResponse({ valid: true, user: rNick7 || rName7, role: rRole7, message: "Passwort gespeichert!" }, callback);
    }

    if (deviceId && deviceId !== "UNKNOWN" && !isWatch && devs7.indexOf(deviceId) === -1) {
      if (devs7.length >= MAX_DEVICES && !isAdmin7) {
        return createJsonResponse({ valid: false, locked: true, limitReached: true, message: "Maximal 3 Geräte erreicht." }, callback);
      }
      if (devs7.length >= MAX_DEVICES && isAdmin7) devs7.shift();
      devs7.push(deviceId);
      sheet.getRange(targetRow + 1, 7).setValue(devs7.join(", "));
    }

    return createJsonResponse({
      valid: true,
      user: rNick7 || rName7,
      realName: rName7,
      role: rRole7,
      unlimitedDevices: isAdmin7
    }, callback);
  }
}

function writeFreeSyncRow(ss, freeKey, serializedJson, nowIso) {
  var sh = ss.getSheetByName("Free_Sync");
  if (!sh) {
    sh = ss.insertSheet("Free_Sync");
    sh.getRange(1, 1, 1, 4).setValues([["Free_ID", "Erstellt_am", "Cloud_Sync_Data", "Last_Sync"]]);
  }
  var data = sh.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0] || "").trim().toUpperCase() === freeKey) {
      sh.getRange(i + 1, 3).setValue(serializedJson);
      sh.getRange(i + 1, 4).setValue(nowIso);
      return;
    }
  }
  sh.appendRow([freeKey, nowIso.slice(0, 10), serializedJson, nowIso]);
}

function analyzePlatePhotoWithGemini(imageBase64, voiceHint) {
  if (!GEMINI_API_KEY) {
    return { status: "no_key", message: "Kein Gemini API-Key hinterlegt." };
  }
  try {
    var cleanB64 = imageBase64.indexOf("base64,") !== -1 ? imageBase64.split("base64,")[1] : imageBase64;
    var url = "https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=" + GEMINI_API_KEY;
    var promptText = "Du bist ein präziser deutscher Sporternährungs-Scanner. Analysiere dieses Essensfoto" +
      (voiceHint ? (" (Hinweis des Nutzers: " + voiceHint + ")") : "") +
      ". Erkenne exakt, was auf dem Teller liegt. Antworte NUR als reines JSON-Objekt im Format: " +
      "{\"mealName\": \"Name des Gerichts auf Deutsch\", \"items\": [{\"name\": \"Zutat\", \"grams\": 150, \"kcal\": 45, \"protein\": 2, \"carbs\": 6, \"fat\": 1}]}";

    var payload = {
      contents: [{
        parts: [
          { text: promptText },
          { inline_data: { mime_type: "image/jpeg", data: cleanB64 } }
        ]
      }],
      generationConfig: { response_mime_type: "application/json" }
    };
    var res = UrlFetchApp.fetch(url, {
      method: "post",
      contentType: "application/json",
      payload: JSON.stringify(payload),
      muteHttpExceptions: true
    });
    var parsed = JSON.parse(res.getContentText());
    var rawJson = parsed.candidates[0].content.parts[0].text;
    return { status: "ok", result: JSON.parse(rawJson) };
  } catch (err) {
    return { status: "error", message: err.toString() };
  }
}
