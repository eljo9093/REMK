// RE:MK GOOGLE APPS SCRIPT BACKEND v28.0 MASTER PRO
// Exakt abgestimmt auf deine Tabelle "RE_MK_Athleten_v28_0_Master_Pro.xlsx":
//
// Reiter "Lizenzen" (Spalten A bis N):
// - Spalte A (1): Name (Klarname, z. B. Jochen Rube, Justin Acikgöz, Balthasar)
// - Spalte B (2): Start-Key (z. B. 1, 1245)
// - Spalte C (3): Status (AKTIV / Gesperrt)
// - Spalte D (4): Rolle / Paket (Admin, Kunde, Trainer)
// - Spalte E (5): Gerät 1 (Smartphone / PC Slot 1)
// - Spalte F (6): Gerät 2 (Smartphone / PC Slot 2)
// - Spalte G (7): Gerät 3 (Smartphone / PC Slot 3) -> Watch zählt NIE ins Limit!
// - Spalte H (8): Nickname (z. B. Admin, CrazyPantz, König, AlexWalz)
// - Spalte I (9): Gültig bis (oder LEBENSLANG)
// - Spalte J (10): Eigenes Passwort (z. B. 26011990, 12345)
// - Spalte K (11): E-Mail / Letzter Login (bleibt zu 100% erhalten!)
// - Spalte L (12): Cloud_Sync_Data (Verlustfreier 2-Wege Live-Sync App <-> Watch)
// - Spalte M (13): Last_Sync (Zeitstempel des letzten Live-Syncs)
// - Spalte N (14): Cloud_Backup_JSON (Komplettes App-Backup & Custom-Library)
// ============================================================================

var MAX_DEVICES = 3;
var ALLOW_FREE_CLOUD_SYNC = true;
var GEMINI_API_KEY = ""; // Optional: Kostenlosen Gemini API-Key von aistudio.google.com hier eintragen

function getLicenseSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  return ss.getSheetByName("Lizenzen") || ss.getSheets()[0];
}

function jsonOut(obj, callback) {
  var txt = JSON.stringify(obj);
  if (callback) {
    return ContentService.createTextOutput(callback + "(" + txt + ")")
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(txt)
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  return handleRemkRequest(e, null);
}

function doPost(e) {
  var body = null;
  try {
    if (e && e.postData && e.postData.contents) {
      body = JSON.parse(e.postData.contents);
    }
  } catch (err) {
    body = null;
  }
  return handleRemkRequest(e, body);
}

function ensureSheetColumnsV28(sheet) {
  var headers = sheet.getRange(1, 1, 1, 14).getValues()[0];
  if (!headers[11]) sheet.getRange(1, 12).setValue("Cloud_Sync_Data");
  if (!headers[12]) sheet.getRange(1, 13).setValue("Last_Sync");
  if (!headers[13]) sheet.getRange(1, 14).setValue("Cloud_Backup_JSON");
}

// Verlustfreier Merge zwischen Smartphone-APK und Pixel Watch 4 Offline-Queue
function mergeCloudSyncPayloads(existingRawStr, incomingPayload) {
  var existingObj = {};
  try {
    if (existingRawStr) existingObj = JSON.parse(existingRawStr);
  } catch (e) {
    existingObj = {};
  }

  var incomingObj = {};
  try {
    incomingObj = (typeof incomingPayload === "string") ? JSON.parse(incomingPayload) : (incomingPayload || {});
  } catch (e) {
    incomingObj = {};
  }

  var merged = {};
  for (var k in existingObj) merged[k] = existingObj[k];
  for (var k2 in incomingObj) merged[k2] = incomingObj[k2];

  // Bewahre immer den aktuellsten Trainings-/Ernährungsplan der Haupt-App für die Uhr
  if (incomingObj.watchPlanCompactJson) {
    merged.watchPlanCompactJson = incomingObj.watchPlanCompactJson;
  } else if (existingObj.watchPlanCompactJson) {
    merged.watchPlanCompactJson = existingObj.watchPlanCompactJson;
  }

  // Höchster Wasserstand des Tages gewinnt verlustfrei
  var oldWater = parseFloat(existingObj.drankLiters || 0) || 0;
  var newWater = parseFloat(incomingObj.drankLiters || 0) || 0;
  merged.drankLiters = Math.max(oldWater, newWater);
  merged.hydrationMl = Math.round(merged.drankLiters * 1000);

  // Merge und dedupliziere alle Watch-Events anhand ihrer eindeutigen Event-ID
  var oldQueue = Array.isArray(existingObj.eventsQueue) ? existingObj.eventsQueue : [];
  var newQueue = Array.isArray(incomingObj.eventsQueue) ? incomingObj.eventsQueue : [];
  var ackIds = Array.isArray(incomingObj.ackEventIds) ? incomingObj.ackEventIds : [];
  var ackMap = {};
  for (var a = 0; a < ackIds.length; a++) ackMap[String(ackIds[a])] = true;

  var combinedMap = {};
  var combinedList = [];
  var allEvents = oldQueue.concat(newQueue);

  for (var i = 0; i < allEvents.length; i++) {
    var ev = allEvents[i];
    if (!ev || !ev.id) continue;
    var evId = String(ev.id);
    if (ackMap[evId]) continue; // Bereits vom Smartphone verarbeitet und bestätigt
    if (!combinedMap[evId]) {
      combinedMap[evId] = true;
      combinedList.push(ev);
    }
  }

  if (combinedList.length > 80) {
    combinedList = combinedList.slice(combinedList.length - 80);
  }
  merged.eventsQueue = combinedList;
  delete merged.ackEventIds;

  return JSON.stringify(merged);
}

// Findet die richtige Athleten-Zeile auch dann zuverlässig, wenn mehrere Nutzer denselben Start-Key ("1") haben
function findMatchingAthleteRow(data, enteredUser, enteredKey) {
  var uLow = String(enteredUser || "").trim().toLowerCase();
  var kLow = String(enteredKey || "").trim().toLowerCase();

  // Durchgang 1: Exakter Treffer auf (Name ODER Nickname) UND (Eigenes Passwort ODER Start-Key)
  if (uLow && kLow) {
    for (var i = 1; i < data.length; i++) {
      var rName = String(data[i][0] || "").trim().toLowerCase();
      var rStartKey = String(data[i][1] || "").trim().replace(/\.0$/, "").toLowerCase();
      var rNick = String(data[i][7] || "").trim().toLowerCase();
      var rCustomPwd = String(data[i][9] || "").trim().replace(/\.0$/, "").toLowerCase();

      var userMatches = (rName === uLow) || (rNick === uLow) || (rName.indexOf(uLow) !== -1) || (uLow.indexOf(rName) !== -1 && rName.length >= 3);
      var keyMatches = (rCustomPwd && rCustomPwd === kLow) || (rStartKey && rStartKey === kLow);
      if (userMatches && keyMatches) return i;
    }
  }

  // Durchgang 2: Eindeutiger Treffer über das persönliche "Eigenes Passwort" (Spalte J)
  if (kLow) {
    for (var j = 1; j < data.length; j++) {
      var cPwd = String(data[j][9] || "").trim().replace(/\.0$/, "").toLowerCase();
      if (cPwd && cPwd === kLow) return j;
    }
  }

  // Durchgang 3: Treffer über eindeutigen Nickname (Spalte H) oder Klarname (Spalte A), falls als Sync-Key auf der Uhr eingegeben
  if (kLow) {
    for (var n = 1; n < data.length; n++) {
      var nickVal = String(data[n][7] || "").trim().toLowerCase();
      var nameVal = String(data[n][0] || "").trim().toLowerCase();
      if ((nickVal && nickVal === kLow) || (nameVal && nameVal === kLow)) return n;
    }
  }

  // Durchgang 4: Treffer über Start-Key (Spalte B)
  if (kLow) {
    for (var s = 1; s < data.length; s++) {
      var sKey = String(data[s][1] || "").trim().replace(/\.0$/, "").toLowerCase();
      if (sKey && sKey === kLow) return s;
    }
  }

  return -1;
}

function handleRemkRequest(e, postBody) {
  try {
    var params = (e && e.parameter) ? e.parameter : {};
    var body = postBody || {};

    var action = String(body.action || params.action || "verify").trim();
    var callback = String(params.callback || "").trim();

    // 1. KI-Teller-Foto-Analyse via Gemini API
    if (action === "analyzePlatePhoto" || action === "analyzeFood") {
      var imgB64 = String(body.imageBase64 || "");
      var hint = String(body.voiceHint || "");
      return jsonOut(analyzePlatePhotoWithGemini(imgB64, hint), callback);
    }

    var enteredUser = String(body.user || params.user || "").trim();
    var enteredKey = String(body.key || body.licenseKey || params.key || "").trim();
    var newKey = String(body.newKey || params.newKey || "").trim();
    var newNickname = String(body.nickname || params.nickname || "").trim();
    var deviceId = String(body.deviceId || params.deviceId || "UNKNOWN").trim();
    var syncPayload = body.syncData || params.syncData || "";

    if (!enteredKey && !enteredUser) {
      return jsonOut({ valid: false, message: "Bitte Lizenz-Key oder Passwort eingeben." }, callback);
    }

    var ss = SpreadsheetApp.getActiveSpreadsheet();

    // 2. Free-User Cloud- & Watch-Sync im eigenen Reiter "Free_Sync"
    if (enteredKey.toUpperCase().indexOf("FREE-") === 0 || enteredKey.toUpperCase() === "FREE_USER") {
      if (!ALLOW_FREE_CLOUD_SYNC) {
        return jsonOut({ valid: false, message: "Cloud-Sync ist nur in der Pro-Version aktiv." }, callback);
      }
      return handleFreeSyncSheet(ss, enteredKey.toUpperCase(), action, syncPayload, callback);
    }

    var sheet = getLicenseSheet();
    ensureSheetColumnsV28(sheet);
    var data = sheet.getDataRange().getValues();
    var targetRow = findMatchingAthleteRow(data, enteredUser, enteredKey);

    if (targetRow === -1) {
      return jsonOut({ valid: false, message: "Ungültiger Name, Key oder Passwort." }, callback);
    }

    var rowNameVal = String(data[targetRow][0] || "Athlet").trim();      // Spalte A: Name
    var rowStartKey = String(data[targetRow][1] || "").trim();           // Spalte B: Start-Key
    var rowStatus = String(data[targetRow][2] || "AKTIV").trim().toLowerCase(); // Spalte C: Status
    var rowRole = String(data[targetRow][3] || "Kunde").trim();          // Spalte D: Rolle / Paket
    var dev1 = String(data[targetRow][4] || "").trim();                  // Spalte E: Gerät 1
    var dev2 = String(data[targetRow][5] || "").trim();                  // Spalte F: Gerät 2
    var dev3 = String(data[targetRow][6] || "").trim();                  // Spalte G: Gerät 3
    var rowNickVal = String(data[targetRow][7] || "").trim();            // Spalte H: Nickname
    var rowExpires = String(data[targetRow][8] || "LEBENSLANG").trim();  // Spalte I: Gültig bis
    var rowCustomPwd = String(data[targetRow][9] || "").trim();          // Spalte J: Eigenes Passwort
    var rowEmailLogin = String(data[targetRow][10] || "").trim();        // Spalte K: E-Mail / Letzter Login
    var rowSyncJson = String(data[targetRow][11] || "").trim();          // Spalte L: Cloud_Sync_Data
    var rowLastSync = String(data[targetRow][12] || "").trim();          // Spalte M: Last_Sync
    var rowBackupJson = String(data[targetRow][13] || "").trim();        // Spalte N: Cloud_Backup_JSON

    if (rowStatus === "gesperrt" || rowStatus === "inaktiv" || rowStatus === "beendet") {
      return jsonOut({ valid: false, locked: true, message: "Zugang ist aktuell pausiert oder gesperrt." }, callback);
    }

    var isAdmin = rowRole.toLowerCase().indexOf("admin") !== -1 ||
                  rowRole.toLowerCase().indexOf("trainer") !== -1 ||
                  rowRole.toLowerCase().indexOf("tsv") !== -1 ||
                  rowNameVal.toLowerCase().indexOf("jochen") !== -1;
    var isWatch = deviceId.toUpperCase().indexOf("WATCH-") === 0;
    var devices = [dev1, dev2, dev3].filter(function(d) { return d && d.length > 0; });

    // AKTION A: Geräte-Plätze (Spalten E, F, G) zurücksetzen
    if (action === "resetDevices") {
      var firstDev = (deviceId && deviceId !== "UNKNOWN" && !isWatch) ? deviceId : "";
      sheet.getRange(targetRow + 1, 5, 1, 3).setValues([[firstDev, "", ""]]);
      SpreadsheetApp.flush();
      return jsonOut({ valid: true, user: rowNickVal || rowNameVal, realName: rowNameVal, role: rowRole, message: "Geräteplätze erfolgreich zurückgesetzt!" }, callback);
    }

    // AKTION B: Erst-Login Nickname in Spalte H speichern
    if (action === "setFirstNickname") {
      var chosenNick = newNickname || enteredUser;
      if (!chosenNick || chosenNick.length < 3) {
        return jsonOut({ valid: false, message: "Bitte wähle einen Nickname mit mindestens 3 Zeichen." }, callback);
      }
      for (var nIdx = 1; nIdx < data.length; nIdx++) {
        if (nIdx === targetRow) continue;
        var otherNick = String(data[nIdx][7] || "").trim().toLowerCase();
        if (otherNick && otherNick === chosenNick.toLowerCase()) {
          return jsonOut({ valid: false, message: "Dieser Nickname ist bereits vergeben." }, callback);
        }
      }
      sheet.getRange(targetRow + 1, 8).setValue(chosenNick);
      SpreadsheetApp.flush();
      return jsonOut({ valid: true, user: chosenNick, realName: rowNameVal, role: rowRole, message: "Nickname gespeichert!" }, callback);
    }

    // AKTION C: Eigenes Passwort in Spalte J speichern (Start-Key in Spalte B bleibt erhalten!)
    if (action === "changePassword" && newKey.length >= 4) {
      sheet.getRange(targetRow + 1, 10).setValue(newKey);
      SpreadsheetApp.flush();
      return jsonOut({ valid: true, user: rowNickVal || rowNameVal, realName: rowNameVal, role: rowRole, message: "Eigenes Passwort in Spalte J gespeichert!" }, callback);
    }

    // AKTION D: Verlustfreier Live-Sync Push (Smartphone <-> Pixel Watch 4) in Spalte L & M
    if (action === "syncPush" && syncPayload) {
      var nowIso = new Date().toISOString();
      var mergedJsonStr = mergeCloudSyncPayloads(rowSyncJson, syncPayload);
      sheet.getRange(targetRow + 1, 12).setValue(mergedJsonStr); // Spalte L: Cloud_Sync_Data
      sheet.getRange(targetRow + 1, 13).setValue(nowIso);        // Spalte M: Last_Sync
      SpreadsheetApp.flush();
      return jsonOut({ valid: true, synced: true, lastSync: nowIso, user: rowNickVal || rowNameVal, role: rowRole }, callback);
    }

    // AKTION E: Live-Sync Pull (liest aus Spalte L & M)
    if (action === "syncPull") {
      var parsedSync = null;
      try { parsedSync = rowSyncJson ? JSON.parse(rowSyncJson) : null; } catch (e) { parsedSync = rowSyncJson; }
      return jsonOut({ valid: true, user: rowNickVal || rowNameVal, realName: rowNameVal, role: rowRole, syncData: parsedSync, lastSync: rowLastSync }, callback);
    }

    // AKTION F: Komplettes Cloud-Backup / Custom-Library speichern & laden (Spalte N)
    if (action === "saveBackup" || action === "syncCustomLibraryV202") {
      var bData = body.backup || body.library || params.backup || "";
      var bStr = (typeof bData === "string") ? bData : JSON.stringify(bData);
      sheet.getRange(targetRow + 1, 14).setValue(bStr); // Spalte N: Cloud_Backup_JSON
      SpreadsheetApp.flush();
      return jsonOut({ valid: true, saved: true }, callback);
    }
    if (action === "loadBackup") {
      return jsonOut({ valid: true, backup: rowBackupJson }, callback);
    }

    // Geräte-Prüfung auf die 3 Einzelspalten E (Gerät 1), F (Gerät 2), G (Gerät 3)
    if (deviceId && deviceId !== "UNKNOWN" && !isWatch) {
      if (devices.indexOf(deviceId) === -1) {
        if (devices.length >= MAX_DEVICES && !isAdmin) {
          return jsonOut({ valid: false, locked: true, limitReached: true, message: "Maximal 3 Geräte erreicht. Bitte Geräte zurücksetzen." }, callback);
        }
        if (devices.length >= MAX_DEVICES && isAdmin) {
          devices.shift();
        }
        devices.push(deviceId);
        sheet.getRange(targetRow + 1, 5, 1, 3).setValues([[
          devices[0] || "",
          devices[1] || "",
          devices[2] || ""
        ]]);
      }
    }

    // Aktualisiere Login-Zeitstempel schonend in Spalte K (E-Mail bleibt erhalten)
    var todayShort = new Date().toISOString().slice(0, 10);
    if (rowEmailLogin.indexOf(todayShort) === -1) {
      var emailPart = rowEmailLogin.split("|")[0].trim();
      var nextColK = (emailPart && emailPart.indexOf("Login:") === -1)
        ? (emailPart + " | Login: " + todayShort)
        : ("Login: " + todayShort);
      sheet.getRange(targetRow + 1, 11).setValue(nextColK);
    }

    SpreadsheetApp.flush();
    var loginSync = null;
    try { loginSync = rowSyncJson ? JSON.parse(rowSyncJson) : null; } catch (e) {}

    return jsonOut({
      valid: true,
      user: rowNickVal || rowNameVal,
      realName: rowNameVal,
      role: rowRole,
      expiresAt: rowExpires,
      needsFirstSetup: (!isAdmin && rowNickVal === ""),
      unlimitedDevices: isAdmin,
      syncData: loginSync,
      lastSync: rowLastSync
    }, callback);

  } catch (err) {
    return jsonOut({ valid: false, message: "Serverfehler: " + err.toString() }, "");
  }
}

function handleFreeSyncSheet(ss, freeKey, action, syncPayload, callback) {
  var sh = ss.getSheetByName("Free_Sync");
  if (!sh) {
    sh = ss.insertSheet("Free_Sync");
    sh.getRange(1, 1, 1, 4).setValues([["Free_ID", "Erstellt_am", "Cloud_Sync_Data", "Last_Sync"]]);
  }
  var data = sh.getDataRange().getValues();
  var rowIdx = -1;
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0] || "").trim().toUpperCase() === freeKey) {
      rowIdx = i;
      break;
    }
  }
  var nowIso = new Date().toISOString();

  if (rowIdx === -1) {
    var initJson = syncPayload ? mergeCloudSyncPayloads("", syncPayload) : "";
    sh.appendRow([freeKey, nowIso.slice(0, 10), initJson, nowIso]);
    SpreadsheetApp.flush();
    return jsonOut({ valid: true, isFreeUser: true, synced: true, syncData: initJson ? JSON.parse(initJson) : null, lastSync: nowIso }, callback);
  }
  var existingRaw = String(data[rowIdx][2] || "").trim();
  if (action === "syncPush" && syncPayload) {
    var mergedFreeStr = mergeCloudSyncPayloads(existingRaw, syncPayload);
    sh.getRange(rowIdx + 1, 3).setValue(mergedFreeStr);
    sh.getRange(rowIdx + 1, 4).setValue(nowIso);
    SpreadsheetApp.flush();
    return jsonOut({ valid: true, isFreeUser: true, synced: true, lastSync: nowIso }, callback);
  }
  var savedObj = null;
  try { savedObj = existingRaw ? JSON.parse(existingRaw) : null; } catch (e) {}
  return jsonOut({ valid: true, isFreeUser: true, syncData: savedObj, lastSync: String(data[rowIdx][3] || "") }, callback);
}

function analyzePlatePhotoWithGemini(imageBase64, voiceHint) {
  if (!GEMINI_API_KEY) {
    return { status: "no_key", message: "Kein Gemini API-Key hinterlegt." };
  }
  try {
    var cleanB64 = imageBase64.indexOf("base64,") !== -1 ? imageBase64.split("base64,")[1] : imageBase64;
    var url = "https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=" + GEMINI_API_KEY;
    var promptText = "Du bist ein präziser deutscher Sporternährungs-Scanner. Analysiere dieses Essensfoto" +
      (voiceHint ? (" (Hinweis des Nutzers: " + voiceHint + ")") : "") +
      ". Erkenne exakt, was auf dem Teller liegt. Antworte NUR als reines JSON-Objekt im Format: " +
      "{\"mealName\": \"Name des Gerichts auf Deutsch\", \"items\": [{\"name\": \"Zutat\", \"grams\": 150, \"kcal\": 45, \"protein\": 2, \"carbs\": 6, \"fat\": 1}]}";

    var payload = {
      contents: [{
        parts: [
          { text: promptText },
          { inline_data: { mime_type: "image/jpeg", data: cleanB64 } }
        ]
      }],
      generationConfig: { response_mime_type: "application/json" }
    };
    var res = UrlFetchApp.fetch(url, {
      method: "post",
      contentType: "application/json",
      payload: JSON.stringify(payload),
      muteHttpExceptions: true
    });
    var parsed = JSON.parse(res.getContentText());
    var rawJson = parsed.candidates[0].content.parts[0].text;
    return { status: "ok", result: JSON.parse(rawJson) };
  } catch (err) {
    return { status: "error", message: err.toString() };
  }
}
